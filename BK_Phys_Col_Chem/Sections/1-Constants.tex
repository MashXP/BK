\subsection{Ideal Gas Constants ($R$)}
\begin{itemize}
    \item $R = 0.0821 \, \mathrm{L \cdot atm \cdot K^{-1} \cdot mol^{-1}}$
    \item $R = 0.0831 \, \mathrm{L \cdot bar \cdot K^{-1} \cdot mol^{-1}}$
    \item $R = 8.314 \times 10^7 \, \mathrm{erg \cdot K^{-1} \cdot mol^{-1}}$
    \item $R = 8.314 \, \mathrm{J \cdot K^{-1} \cdot mol^{-1}} = 8.314 \, \mathrm{kPa \cdot dm^3 \cdot K^{-1} \cdot mol^{-1}}$
    \item $R = 1.99 \, \mathrm{cal \cdot K^{-1} \cdot mol^{-1}}$
\end{itemize}

\subsection{Other Constants}
\begin{itemize}
    \item \textbf{Vacuum Permittivity:} $\epsilon_0 = 8.8542 \times 10^{-12} \, \mathrm{C^2 \cdot N^{-1} \cdot m^{-2}}$ (or F/m)
    \item \textbf{Elementary Charge:} $q = 1.6 \times 10^{-19} \, \mathrm{C}$
    \item \textbf{Boltzmann Constant:} $k = 1.3806 \times 10^{-23} \, \mathrm{J/K} = 1.3806 \times 10^{-16} \, \mathrm{erg/K}$
    \item \textbf{Avogadro Constant:} $N_A = 6.022 \times 10^{23} \, \mathrm{mol^{-1}}$
    \item \textbf{Faraday Constant:} $F = 96500 \, \mathrm{C/mol}$
    \item \textbf{Debye-Hückel constant (Water, 25$^\circ$C):} $A = 0.509 \, (\mathrm{mol/L})^{-0.5}$
    \item \textbf{Ramsay-Shields constant:} $k = 2.1 \, \mathrm{erg/K} = 2.1 \times 10^{-7} \, \mathrm{J/K}$
    \item \textbf{Gravity:} $g = 9.81 \, \mathrm{m/s^2}$
\end{itemize}
