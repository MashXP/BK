\subsection{Diffusion \& Brownian Motion}
\textbf{Mean Square Displacement:}
\begin{equation}
    \langle x^2 \rangle = 2Dt \implies \langle x \rangle = \sqrt{2Dt}
\end{equation}
\textbf{Fick's First Law:}
\begin{equation}
    J = \frac{1}{S} \frac{dn}{dt} = -D \frac{dC}{dx}
\end{equation}
\textbf{Einstein-Stokes Equation:}
\begin{equation}
    D = \frac{kT}{f} = \frac{kT}{6 \pi \eta r} = \frac{RT}{6 \pi \eta r N_A}
\end{equation}

\subsection{Sedimentation}
\textbf{Terminal Velocity ($v_t$) in Gravity:}
\begin{equation}
    v_t = \frac{2 r^2 g (\rho_p - \rho_{fluid})}{9 \eta}
\end{equation}
\textbf{In Centrifugal Field:}
\begin{equation}
    v_t = \frac{dr}{dt} = \frac{d^2 (\rho_p - \rho) r \omega^2}{18 \eta}
\end{equation}
\textbf{Sedimentation Coefficient ($s$):}
\begin{equation}
    s = \frac{v_t}{r \omega^2} = \frac{\ln(r_2/r_1)}{\omega^2 (t_2 - t_1)} = \frac{m(1-\bar{v}\rho)}{f}
\end{equation}

\subsection{Osmotic Pressure ($\Pi$)}
For polymers/colloids (virial expansion):
\begin{equation}
    \Pi = c RT \left( \frac{1}{M} + B_2 c + B_3 c^2 + \dots \right)
\end{equation}
Plot $\Pi/c$ vs. $c$: intercept = $RT/M$.

\subsection{Viscosity ($\eta$)}
\textbf{Einstein's Equation (for dilute spheres):}
\begin{equation}
    \eta = \eta_0 (1 + 2.5 \phi)
\end{equation}
where $\phi$ is the volume fraction of particles.
