\section{Second and Third Laws of Thermodynamics}

\subsection{Entropy ($S$) and Spontaneity}
\begin{itemize}
    \item \textbf{Entropy}: A state function measuring the degree of disorder or probability of a state.
    \item \textbf{Second Law}: For any process, $dS \geq \frac{\delta Q}{T}$.
    \item \textbf{Entropy Balance}: $S_2 - S_1 = \int_1^2 \frac{\delta Q}{T} + S_{gen}$, where $S_{gen} \geq 0$ (0 for reversible, $>0$ for irreversible).
    \item \textbf{Isolated Systems}: $\Delta S \geq 0$. Spontaneous processes proceed toward maximum entropy.
    \item \textbf{Universe}: $\Delta S_{univ} = \Delta S_{sys} + \Delta S_{surr} \geq 0$.
\end{itemize}

\subsection{Entropy Calculations}
\begin{itemize}
    \item \textbf{Phase Change}: $\Delta S = \frac{\Delta H_{phase}}{T_{phase}}$.
    \item \textbf{Heating/Cooling}: $\Delta S = \int_{T_1}^{T_2} \frac{C}{T} dT$.
    \item \textbf{Isothermal (Ideal Gas)}: $\Delta S = nR \ln \frac{V_2}{V_1} = nR \ln \frac{P_1}{P_2}$.
    \item \textbf{Standard Reaction Entropy}: $\Delta S^0_{\text{rxn}} = \sum n S^0_{\text{products}} - \sum n S^0_{\text{reactants}}$.
\end{itemize}

\subsection{Gibbs Free Energy ($G$)}
\begin{itemize}
    \item \textbf{Definition}: $G = H - TS$. At constant $T$: $\Delta G = \Delta H - T \Delta S$.
    \item \textbf{Spontaneity Criteria} (Const $T, P$):
    \begin{itemize}
        \item $\Delta G < 0$: Spontaneous forward reaction.
        \item $\Delta G = 0$: System at equilibrium.
        \item $\Delta G > 0$: Non-spontaneous (spontaneous in reverse).
    \end{itemize}
    \item \textbf{Standard Gibbs Energy}: \\
    $\Delta G^0_{\text{rxn}} = \sum n \Delta G^0_{\text{f,products}} - \sum n \Delta G^0_{\text{f,reactants}}$.
\end{itemize}

\subsection{Third Law of Thermodynamics}
\textbf{Planck Statement}: The entropy of a pure, perfectly crystalline substance at absolute zero is zero: $\lim_{T \to 0} S = 0$. Values of $S$ at $T > 0$ are "absolute entropies".
