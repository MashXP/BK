\section{Vapor-Liquid and Liquid-Liquid Equilibrium}

\subsection{Solutions and Chemical Potential}
\begin{itemize}
    \item \textbf{Ideal Solution}: $\mu_i = \mu_i^*(T) + RT \ln x_i$.
    \item \textbf{Real Solution}: $\mu_i = \mu_i^*(T) + RT \ln a_i$ (where $a_i$ is activity).
\end{itemize}

\subsection{Solubility of Gases in Liquids}
\begin{itemize}
    \item \textbf{Henry's Law} (Infinitely dilute): $x_i = k_H P_i$.
    \item \textbf{Sreder Equation} (Effect of $T$):
    \begin{equation}
        \left( \frac{\partial \ln x_i}{\partial T} \right)_P = \frac{\lambda_i}{RT^2}
    \end{equation}
\end{itemize}

\subsection{Vapor-Liquid Equilibrium (VLE)}
\begin{itemize}
    \item \textbf{Raoult's Law}: $P_i = P_i^0 x_i^l$.
    \item \textbf{Konovalov I}: $x_B^h = \frac{\alpha x_B^l}{1 + (\alpha - 1)x_B^l}$ where $\alpha = P_B^0 / P_A^0$.
    \item \textbf{Lever Rule}: 
    \begin{equation}
        \frac{n_{\text{liquid}}}{n_{\text{vapor}}} = \frac{x_B^h - x_B}{x_B - x_B^l}
    \end{equation}
\end{itemize}

\subsection{Steam Distillation}
For immiscible liquids (e.g., water and oil):
\begin{itemize}
    \item Boiling occurs when $P_{H_2O}^0 + P_A^0 = P_{\text{ext}}$.
    \item \textbf{Mass of steam required} to extract 1 kg of A:
    \begin{equation}
        \frac{g_{H_2O}}{g_A} = \frac{P_{H_2O}^0 \cdot 18}{P_A^0 \cdot M_A}
    \end{equation}
\end{itemize}

\subsection{Liquid-Liquid Equilibrium (LLE)}
\begin{itemize}
    \item \textbf{Partially Miscible}: Solubility varies with $T$. Upper/Lower Critical Solution Temperature (UCST/LCST).
    \item \textbf{Ternary Systems} ($k=3$): Represented on triangular diagrams.
    \begin{equation}
        x_A + x_B + x_C = 1
    \end{equation}
\end{itemize}
