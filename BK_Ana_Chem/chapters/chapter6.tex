\subsection*{Chapter 6: Spectroscopic Methods}

\textbf{UV} : 190 - 380nm
\textbf{Vis} : 380 - 780nm  

\textbf{Transmittance (T) and Absorbance (A):}
\begin{align}
T &= \frac{P_T}{P_0} \\
A &= -\log T = \log \frac{P_0}{P_T} \\
A &= 2.00 - \log \%T
\end{align}
\textit{P\textsubscript{T}}: Power of transmitted radiation, \textit{P\textsubscript{0}}: Power of incident radiation.

\vspace{5mm}

\textbf{Lambert-Beer's Law:}
\begin{equation}
A = \epsilon b C
\end{equation}
\textit{A}: Absorbance (unitless), \textit{\(\epsilon\)}: Molar absorptivity (L·mol\(^{-1}\)·cm\(^{-1}\)), \textit{b}: Path length (cm), \textit{C}: Concentration (mol/L).
\vspace{5mm}

\textbf{Multicomponent Samples:}
\begin{equation}
A_{\text{tot}} = A_X + A_Y = \epsilon_X b C_X + \epsilon_Y b C_Y
\end{equation}
