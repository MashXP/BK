\section{METHOD}

\subsection{Preparation}
\begin{enumerate}
    \item Close all liquid valves.
    \item Turn on the air valve and switch on the air blower for 5 minutes to remove moisture from the column. Then, shut down the blower.
    \item Open valve "1" and valve "3", and pump water into the bottom part of the column.
    \item Use valve "2" to adjust the water level to equal the height of the level device (approximately 3/4 of the bottom part).
    \item Turn off the pump and close valve "3".
\end{enumerate}

\subsection{Measurement of pressure drop for dried column}
\begin{enumerate}
    \item Ensure all liquid valves are closed.
    \item Open air valve "6" while valve "5" is closed.
    \item Turn on the air blower and open valve "5" slowly to adjust the gas flow.
    \item For a fixed gas flow rate, determine $\Delta P_{dry}$ on the pressure gauge (in $mmH_2O$).
    \item Turn off the blower and wait for 5 minutes between measurements.
\end{enumerate}

\subsection{Measurement for pressure drop of the wetted column}
\begin{enumerate}
    \item Switch on the air blower and adjust gas flow rate to 15-20\% of maximum.
    \item Open liquid valve "1" and turn on the pump.
    \item Use valve "$V_L$" from the liquid flowmeter to adjust the water flow rate.
    \item At a specific liquid flow rate, alter the air flow rate and measure $\Delta P_{wet}$ until the column reaches the flooding point.
    \item Repeat the process for different liquid flow rates.
\end{enumerate}

\textbf{Note:} Maintain the water level at 3/4 of the bottom part throughout the measurement using valve "4" or "2" as needed.
