\section{ABSTRACT}

\subsection{Experimental purpose}
\begin{itemize}
    \item Survey retention time of a system of continuous stirred-tank reactors (CSTRs).
    \item Determine real retention time distribution function and compare with theoretical retention time distribution function.
\end{itemize}

\subsection{Experimental results}
\begin{itemize}
    \item Maximum absorbance (for 1 reactor): $D_0 = 0.03$
    \item Reactor diameter: $d = 14.8$ (cm)
    \item Water level in the CSTR: $H = 12$ (cm)
    \item Flowrate of water: $v = 0.4$ (liter/min)
\end{itemize}
